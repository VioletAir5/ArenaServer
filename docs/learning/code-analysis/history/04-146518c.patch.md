+#include "net/PacketEcho.h"
+
+#include <exception>
+#include <iostream>
+#include <memory>
+#include <stdexcept>
+
+#include "protocol/PacketCodec.hpp"
+
+namespace service::gateway
+{
+ConnectionMessageCallback
+makePacketEchoHandler(std::size_t maxBodySize)
+{
+    if (maxBodySize > kMaxPacketSize)
+    {
+        throw std::invalid_argument(
+            "gateway packet body limit exceeds 256 KiB"
+        );
+    }
+    auto codec = std::make_shared<protocol::PacketCodec>(maxBodySize);
+    return [codec](
+               IConnection& connection,
+               std::span<const std::byte> bytes
+           )
+    {
+        try
+        {
+            const auto status = codec->feed(
+                bytes,
+                [&](const protocol::PacketHeader& request,
+                    std::string_view payload)
+                {
+                    if (request.m_type !=
+                        protocol::PacketHeader::kEchoRequest)
+                    {
+                        connection.close();
+                        return false;
+                    }
+                    auto response = protocol::PacketCodec::encode(
+                        protocol::PacketHeader::kEchoResponse,
+                        request.m_requestId,
+                        std::as_bytes(std::span(payload))
+                    );
+                    if (!response)
+                    {
+                        connection.close();
+                        return false;
+                    }
+                    connection.send(std::as_bytes(std::span(
+                        response->peek(),
+                        response->readableBytes()
+                    )));
+                    return connection.connected();
+                }
+            );
+            if (status == protocol::FrameDecodeStatus::Invalid)
+            {
+                std::cerr << "PacketCodec invalid frame id="
+                          << connection.id() << '\n';
+                connection.close();
+            }
+        }
+        catch (const std::exception& error)
+        {
+            std::cerr << "PacketCodec callback failed: "
+                      << error.what() << '\n';
+            connection.close();
+        }
+        catch (...)
+        {
+            connection.close();
+        }
+    };
+}
+} // namespace service::gateway

+std::size_t checkedLimit(std::size_t limit)
+{
+    if (limit < PacketHeader::kMetadataSize ||
+        limit > PacketHeader::kMaxPacketSize)
+    {
+        throw std::invalid_argument("invalid packet body limit");
+    }
+    return limit;
+}

+PacketCodec::PacketCodec(std::size_t limit)
+    : m_maxBodySize(checkedLimit(limit))
+    , m_input(
+          std::min<std::size_t>(
+              4096, limit + PacketHeader::kLengthSize
+          ),
+          limit + PacketHeader::kLengthSize,
+          0
+      )
+{
+}
+
+FrameDecodeStatus PacketCodec::feed(
+    std::span<const std::byte> bytes,
+    const PacketHeaderCodec::FrameCallback& onFrame
+)
+{
+    if (m_feeding)
+    {
+        throw std::logic_error("reentrant PacketCodec::feed");
+    }
+    if (m_status != FrameDecodeStatus::NeedMore)
+    {
+        return m_status;
+    }
+    m_feeding = true;
+    struct Guard
+    {
+        bool& flag;
+        ~Guard()
+        {
+            flag = false;
+        }
+    } guard{m_feeding};
+    try
+    {
+        for (;;)
+        {
+            const auto size = m_input.readableBytes();
+            std::size_t target = PacketHeader::kLengthSize;
+            if (size >= PacketHeader::kLengthSize)
+            {
+                PacketHeader header;
+                auto status = PacketHeaderCodec::decode(
+                    m_input.peek(),
+                    size,
+                    header
+                );
+                std::uint32_t networkLength;
+                std::memcpy(
+                    &networkLength,
+                    m_input.peek(),
+                    sizeof(networkLength)
+                );
+                const auto bodyLength = ::ntohl(networkLength);
+                if (status == HeaderDecodeStatus::Invalid ||
+                    bodyLength > m_maxBodySize)
+                {
+                    return m_status = FrameDecodeStatus::Invalid;
+                }
+                target = size < PacketHeader::kHeaderSize
+                             ? PacketHeader::kHeaderSize
+                             : PacketHeader::kLengthSize + bodyLength;
+                if (size == target &&
+                    status == HeaderDecodeStatus::Complete)
+                {
+                    auto result = PacketHeaderCodec::decodeFrames(
+                        m_input,
+                        onFrame
+                    );
+                    if (result != FrameDecodeStatus::NeedMore)
+                    {
+                        return m_status = result;
+                    }
+                    continue;
+                }
+            }
+            if (bytes.empty())
+            {
+                return FrameDecodeStatus::NeedMore;
+            }
+            const auto count = std::min(target - size, bytes.size());
+            if (!m_input.append(bytes.data(), count))
+            {
+                return m_status = FrameDecodeStatus::Invalid;
+            }
+            bytes = bytes.subspan(count);
+        }
+    }
+    catch (...)
+    {
+        m_status = FrameDecodeStatus::Invalid;
+        throw;
+    }
+}
+
+std::size_t PacketCodec::pendingBytes() const noexcept
+{
+    return m_input.readableBytes();
+}
+
+std::optional<Buffer> PacketCodec::encode(
+    std::uint16_t type,
+    std::uint32_t requestId,
+    std::span<const std::byte> payload,
+    std::uint8_t version
+)
+{
+    if (payload.size() >
+        PacketHeader::kMaxPacketSize - PacketHeader::kMetadataSize)
+    {
+        return std::nullopt;
+    }
+    PacketHeader header;
+    header.m_length = PacketHeader::kMetadataSize + payload.size();
+    header.m_type = type;
+    header.m_requestId = requestId;
+    header.m_version = version;
+    const auto wireSize = PacketHeader::kHeaderSize + payload.size();
+    Buffer output(std::min<std::size_t>(4096, wireSize), wireSize);
+    if (!output.append(payload.data(), payload.size()) ||
+        !PacketHeaderCodec::encode(output, header))
+    {
+        return std::nullopt;
+    }
+    return output;
+}

+def packet(payload, request_id=1, packet_type=1, version=1):
+    return struct.pack("!IHIB", len(payload) + 7, packet_type, request_id, version) + payload
+
+def receive_exact(peer, length):
+    result = bytearray()
+    while len(result) < length:
+        chunk = peer.recv(length - len(result))
+        if not chunk:
+            raise AssertionError("unexpected EOF inside response")
+        result.extend(chunk)
+    return bytes(result)
+
+def expect_closed(peer):
+    try:
+        if peer.recv(1) != b"":
+            raise AssertionError("invalid/truncated frame produced a response")
+    except ConnectionResetError:
+        pass
+

+namespace

+std::string frame(std::string_view payload, std::uint32_t id = 1)
+{
+    auto encoded = PacketCodec::encode(
+        PacketHeader::kEchoRequest,
+        id,
+        std::as_bytes(std::span(payload))
+    );
+    if (!encoded)
+    {
+        throw std::runtime_error("encode");
+    }
+    return {encoded->peek(), encoded->readableBytes()};
+}
+auto bytes(std::string_view data)
+{
+    return std::as_bytes(std::span(data));
+}
+} // namespace

+TEST(PacketCodecTest, DecodesBytewiseHeadersAndCoalescedBinaryFrames)
+{
+    const std::string binary{"a\0b\xff", 4};
+    const auto first = frame(binary, 0x01020304);
+    const auto next = frame("", 42) + frame("tail", 43);
+    PacketCodec codec;
+    std::vector<std::string> payloads;
+    std::vector<std::uint32_t> ids;
+    auto receive =
+        [&](const PacketHeader& header, std::string_view payload)
+    {
+        ids.push_back(header.m_requestId);
+        payloads.emplace_back(payload);
+        return true;
+    };
+    for (char byte : first)
+    {
+        EXPECT_EQ(
+            codec.feed(bytes(std::string_view(&byte, 1)), receive),
+            FrameDecodeStatus::NeedMore
+        );
+    }
+    EXPECT_EQ(
+        codec.feed(bytes(next), receive),
+        FrameDecodeStatus::NeedMore
+    );
+    EXPECT_EQ(
+        (std::vector<std::string>{binary, "", "tail"}),
+        payloads
+    );
+    EXPECT_EQ((std::vector<std::uint32_t>{0x01020304, 42, 43}), ids);
+    EXPECT_EQ(codec.pendingBytes(), 0);
+}
