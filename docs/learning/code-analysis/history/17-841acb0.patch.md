+namespace service::gateway
+{
+TokenBucketFeature::TokenBucketFeature(
+    double capacity, double rate, FeatureResult exhausted, Now now
+)
+    : m_capacity(capacity)
+    , m_rate(rate)
+    , m_exhausted(exhausted)
+    , m_now(std::move(now))
+    , m_tokens(capacity)
+{
+    if (!std::isfinite(capacity) || capacity < 1 ||
+        !std::isfinite(rate) || rate <= 0 || !m_now ||
+        (exhausted != FeatureResult::Drop &&
+         exhausted != FeatureResult::Close))
+        throw std::invalid_argument("invalid token bucket");
+}
+void TokenBucketFeature::onConnected(IConnectionControl&)
+{
+    m_last = m_now();
+}
+FeatureResult
+TokenBucketFeature::onPacket(IConnectionControl& control, PacketView)
+{
+    auto now = m_now();
+    if (now > m_last)
+    {
+        const double elapsed = std::chrono::duration<double>(
+                                   now - m_last
+        )
+                                   .count();
+        m_tokens = std::min(m_capacity, m_tokens + elapsed * m_rate);
+        m_last = now;
+    }
+    if (m_tokens >= 1)
+    {
+        m_tokens -= 1;
+        return FeatureResult::Continue;
+    }
+    if (m_exhausted == FeatureResult::Close)
+        std::cerr << "connection packet rate exceeded id="
+                  << control.context().connectionId << '\n';
+    return m_exhausted;
+}

+def check(port, ws=False):
+    def connect():
+        if ws:
+            peer, response = upgrade(port)
+            assert response.startswith(b"HTTP/1.1 101")
+            return peer
+        peer = socket.create_connection(("127.0.0.1", port), 5)
+        peer.settimeout(5)
+        return peer
+
+    def send(peer, data):
+        if ws:
+            send_frame(peer, 2, data)
+        else:
+            peer.sendall(data)
+
+    def echo(peer, payload):
+        if ws:
+            assert read_frame(peer) == (2, packet(payload, 2))
+        else:
+            assert read_tcp_packet(peer) == packet(payload, 2)
+
+    def closed(peer):
+        if ws:
+            assert close_code(peer) == 1008
+        else:
+            try:
+                assert peer.recv(1) == b"", "rejected packet reached business"
+            except ConnectionResetError:
+                pass
+
+    with connect() as first, connect() as independent:
+        # PacketCodec fragments can cross TCP reads or WS messages;
+        # their count must not consume packet tokens.
+        data = packet(b"fragmented")
+        send(first, data[:5])
+        send(independent, packet(b"independent"))
+        echo(independent, b"independent")
+        send(first, data[5:])
+        echo(first, b"fragmented")
+        send(first, packet(b"second"))
+        echo(first, b"second")
+        send(first, packet(b"excess"))
+        closed(first)
+        send(independent, packet(b"second independent"))
+        echo(independent, b"second independent")
+
+    with connect() as coalesced:
+        send(coalesced, packet(b"one") + packet(b"two"))
+        echo(coalesced, b"one")
+        echo(coalesced, b"two")
+        send(coalesced, packet(b"three"))
+        closed(coalesced)

+    with connect() as denied:
+        send(denied, packet(b"disallowed", 2))
