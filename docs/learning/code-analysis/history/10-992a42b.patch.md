+// A context exists for one complete frame. payload refers to PacketCodec's
+// input and must not be retained after the callback returns.
+struct PacketContext
+{
+    IConnection& connection;
+    const protocol::PacketHeader& header;
+    std::string_view payload;
+    std::optional<arena::protocol::Envelope> envelope{};
+};
+
+enum class PacketStageResult
+{
+    Continue, // Run the next stage.
+    Handled,  // Stop this chain; accept the current frame.
+    Reject    // Stop this chain and close the connection.
+};
+
+// Configured once per connection, run serially by that connection's owner.
+// Framing stays in PacketCodec; stages only see complete frames.
+class PacketPipeline
+{
+public:
+    using Stage = std::function<PacketStageResult(PacketContext&)>;
+
+    void addStage(Stage stage);
+    PacketStageResult process(PacketContext& context) const;
+
+    // Opt-in stage for protobuf payloads. Later stages can read
+    // context.envelope; malformed or over-limit payloads are rejected.
+    static PacketStageResult decodeEnvelope(PacketContext& context);
+
+private:
+    std::vector<Stage> m_stages;
+};
+
+// Creates a per-connection decoder and attaches the ordered pipeline to
+// complete-frame callbacks. Reject closes the connection.
+ConnectionMessageCallback makePacketHandler(
+    PacketPipeline pipeline, std::size_t maxBodySize
+);
+} // namespace service::gateway

+#include "net/PacketPipeline.h"

+#include <exception>
+#include <iostream>

+#include <stdexcept>
+#include <utility>

+#include "protocol/EnvelopeCodec.hpp"
+#include "protocol/PacketCodec.hpp"
+
+namespace service::gateway
+{
+void PacketPipeline::addStage(Stage stage)
+{
+    if (!stage)
+    {
+        throw std::invalid_argument("empty packet stage");
+    }
+    m_stages.push_back(std::move(stage));
+}
+
+PacketStageResult PacketPipeline::process(PacketContext& context) const
+{
+    for (const auto& stage : m_stages)
+    {
+        const auto result = stage(context);
+        if (result != PacketStageResult::Continue)
+        {
+            return result;
+        }
+    }
+    return PacketStageResult::Continue;
+}
+
+PacketStageResult PacketPipeline::decodeEnvelope(PacketContext& context)
+{
+    arena::protocol::Envelope decoded;
+    if (!protocol::EnvelopeCodec::decode(
+            std::as_bytes(std::span(context.payload)), decoded))
+    {
+        return PacketStageResult::Reject;
+    }
+    context.envelope.emplace(std::move(decoded));
+    return PacketStageResult::Continue;
+}
+
+ConnectionMessageCallback makePacketHandler(
+    PacketPipeline pipeline, std::size_t maxBodySize
+)
+{
+    if (maxBodySize > kMaxPacketSize)
+    {
+        throw std::invalid_argument(
+            "gateway packet body limit exceeds 256 KiB"
+        );
+    }
+    auto codec = std::make_shared<protocol::PacketCodec>(maxBodySize);
+    return [codec, pipeline = std::move(pipeline)](
+               IConnection& connection,
+               std::span<const std::byte> bytes
+           )
+    {
+        try
+        {
+            const auto status = codec->feed(
+                bytes,
+                [&](const protocol::PacketHeader& header,
+                    std::string_view payload)
+                {
+                    PacketContext context{connection, header, payload};
+                    if (pipeline.process(context) ==
+                        PacketStageResult::Reject)
+                    {
+                        connection.close();
+                        return false;
+                    }
+                    return connection.connected();
+                }
+            );
+            if (status == protocol::FrameDecodeStatus::Invalid)
+            {
+                std::cerr << "PacketCodec invalid frame id="
+                          << connection.id() << '\n';
+                connection.close();
+            }
+        }
+        catch (const std::exception& error)
+        {
+            std::cerr << "PacketPipeline callback failed: "
+                      << error.what() << '\n';
+            connection.close();
+        }
+        catch (...)
+        {
+            connection.close();
+        }
+    };
+}
+} // namespace service::gateway

+std::string frame(std::string_view payload)
+{
+    auto encoded = service::protocol::PacketCodec::encode(
+        service::protocol::PacketHeader::kEchoRequest, 7, bytes(payload));
+    if (!encoded)
+    {
+        throw std::runtime_error("encode failed");
+    }
+    return {encoded->peek(), encoded->readableBytes()};
+}

+TEST(PacketPipelineTest, RunsInOrderAndShortCircuitsHandledOrReject)
+{
+    TestConnection connection;
+    service::protocol::PacketHeader header;
+    PacketContext context{connection, header, "payload"};
+    PacketPipeline pipeline;
+    std::vector<int> order;
+    pipeline.addStage([&](PacketContext& packet) {
+        EXPECT_EQ(packet.payload, "payload");
+        order.push_back(1);
+        return PacketStageResult::Continue;
+    });
+    pipeline.addStage([&](PacketContext&) {
+        order.push_back(2);
+        return PacketStageResult::Handled;
+    });
+    pipeline.addStage([&](PacketContext&) {
+        order.push_back(3);
+        return PacketStageResult::Continue;
+    });
+    EXPECT_EQ(pipeline.process(context), PacketStageResult::Handled);
+    EXPECT_EQ(order, (std::vector<int>{1, 2}));
+
+    PacketPipeline rejecting;
+    rejecting.addStage([](PacketContext&) {
+        return PacketStageResult::Reject;
+    });
+    rejecting.addStage([&](PacketContext&) {
+        order.push_back(4);
+        return PacketStageResult::Continue;
+    });
+    EXPECT_EQ(rejecting.process(context), PacketStageResult::Reject);
+    EXPECT_EQ(order, (std::vector<int>{1, 2}));
+    EXPECT_THROW(pipeline.addStage({}), std::invalid_argument);
+}

+TEST(PacketPipelineTest, EnvelopeDecodeStageFeedsLaterStagesAndRejectsMalformed)
+{
+    arena::protocol::Envelope envelope;
+    envelope.mutable_ping()->set_nonce(123);
+    const auto wire = service::protocol::EnvelopeCodec::encode(envelope);
+    ASSERT_TRUE(wire);
+    TestConnection connection;
+    service::protocol::PacketHeader header;
+    PacketPipeline pipeline;
+    pipeline.addStage(PacketPipeline::decodeEnvelope);
+    int routed = 0;
+    pipeline.addStage([&](PacketContext& packet) {
+        EXPECT_TRUE(packet.envelope.has_value());
+        EXPECT_EQ(packet.envelope->ping().nonce(), 123);
+        ++routed;
+        return PacketStageResult::Handled;
+    });
+    PacketContext valid{connection, header, *wire};
+    EXPECT_EQ(pipeline.process(valid), PacketStageResult::Handled);
+    EXPECT_EQ(routed, 1);
+    PacketContext malformed{
+        connection, header, std::string_view("\x62\x80", 2)
+    };
+    EXPECT_EQ(pipeline.process(malformed), PacketStageResult::Reject);
+    EXPECT_FALSE(malformed.envelope.has_value());
+    EXPECT_EQ(routed, 1);
+}
