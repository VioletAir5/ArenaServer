+## TCP and WebSocket pre-login handshake
+
+`ArenaServer asio handshake [ws-origin]` listens on TCP `127.0.0.1:8888` and WebSocket `127.0.0.1:8889/ws`. The default allowed Origin is `http://localhost:3000`; a different or missing Origin receives HTTP 403. Existing `raw` and `packet` modes retain their behavior. The `handshake` mode is available with the Asio backend.
+
+Both transports carry the same binary PacketCodec frame (magic, length, type, request ID, version, payload). A pre-login request uses type 1 and a protobuf Envelope containing Ping; the server returns type 2 with Envelope Pong and the same nonce. This verifies transport and protocol readiness before login. Authentication and LoginRequest handling are not implemented yet.
+
+WebSocket requires binary messages, limits each message to 256 KiB plus the 8-byte PacketCodec prefix, and caps queued outbound bytes at 1 MiB per connection. Invalid application packets close with code 1008; text messages close with 1003; oversized messages close with 1009. Local and peer close codes are written to the server log. Output overflow closes the connection, with a five-second fallback if the peer does not finish the close handshake.

+#include <span>
+#include <utility>

+#include "net/PacketPipeline.h"
+#include "protocol/EnvelopeCodec.hpp"
+#include "protocol/PacketCodec.hpp"

+namespace service::gateway

+ConnectionMessageCallback makePreLoginHandshakeHandler()
+{
+    PacketPipeline pipeline;
+    pipeline.addStage(PacketPipeline::decodeEnvelope);
+    pipeline.addStage(
+        [](PacketContext& context)
+        {
+            if (context.header.m_type !=
+                    protocol::PacketHeader::kEchoRequest ||
+                !context.envelope->has_ping())
+            {
+                return PacketStageResult::Reject;
+            }
+            arena::protocol::Envelope reply;
+            reply.set_request_id(context.envelope->request_id());
+            reply.mutable_pong()->set_nonce(
+                context.envelope->ping().nonce()
+            );
+            const auto payload = protocol::EnvelopeCodec::encode(reply
+            );
+            if (!payload)
+            {
+                return PacketStageResult::Reject;
+            }
+            auto frame = protocol::PacketCodec::encode(
+                protocol::PacketHeader::kEchoResponse,
+                context.header.m_requestId,
+                std::as_bytes(std::span(*payload))
+            );
+            if (!frame)
+            {
+                return PacketStageResult::Reject;
+            }
+            context.connection.send(std::as_bytes(
+                std::span(frame->peek(), frame->readableBytes())
+            ));
+            return context.connection.connected()
+                       ? PacketStageResult::Handled
+                       : PacketStageResult::Reject;
+        }
+    );
+    return makePacketHandler(std::move(pipeline), kMaxPacketSize);
+}
+} // namespace service::gateway

+        acceptor.listen();
+        acceptNext();
+    }
+
+    void acceptNext()
+    {
+        if (stopping)
+            return;
+        auto self = shared_from_this();
+        acceptor.async_accept(
+            [self](Error error, Tcp::socket socket)
+            {
+                if (!error && !self->stopping)
+                {
+                    if (self->connections.size() >= self->capacity)
+                    {
+                        Error ignored;
+                        socket.close(ignored);
+                    }
+                    else
+                    {
+                        try
+                        {
+                            const auto id = detail::nextConnectionId(
+                            );
+                            auto handler = self->factory();
+                            if (!handler)
+                                throw std::invalid_argument(
+                                    "empty handler"
+                                );
+                            auto session = std::make_shared<
+                                WebSocketConnection>(
+                                self->context,
+                                std::move(socket),
+                                id,
+                                self->allowedOrigin,
+                                std::move(handler),
+                                [weak = self->weak_from_this(
+                                 )](std::uint64_t key)
+                                {
+                                    if (auto owner = weak.lock())
+                                        owner->connections.erase(key);
+                                }
+                            );
+                            self->connections.emplace(id, session);
+                            session->start();
+                        }
+                        catch (const std::exception& exception)
+                        {
+                            std::cerr << "WebSocket setup: "
+                                      << exception.what() << '\n';
+                        }
+                    }
+                }
+                if (!self->stopping)
+                    self->acceptNext();
+            }
+        );
+    }
+
+    void stop() noexcept
+    {
+        if (stopping)
+            return;
+        stopping = true;
+        Error ignored;
+        acceptor.close(ignored);
+        auto pending = std::move(connections);
+        connections.clear();
+        for (auto& [id, session] : pending)
+        {
+            (void)id;
+            session->abort();
+        }
+    }
+};
+
+WebSocketServer::WebSocketServer(
+    asio::io_context& context,
+    const InetAddress& address,
+    std::string allowedOrigin,
+    ConnectionHandlerFactory handlerFactory,
+    std::size_t maxConnections
+)
+    : m_impl(std::make_shared<Impl>(
+          context,
+          std::move(allowedOrigin),
+          std::move(handlerFactory),
+          maxConnections
+      ))
+{
+    m_impl->start(address);
+}
+WebSocketServer::~WebSocketServer() noexcept
+{
+    m_impl->stop();
+}
+void WebSocketServer::stop() noexcept
+{
+    m_impl->stop();
+}
+std::size_t WebSocketServer::connectionCount() const noexcept
+{
+    return m_impl->connections.size();
+}

+import base64
+import os
+import re

+import struct

+MAGIC = 0x4152454E
+ORIGIN = "http://localhost:3000"
+
+
+def exact(peer, count):
+    data = bytearray()
+    while len(data) < count:
+        chunk = peer.recv(count - len(data))
+        if not chunk:
+            raise AssertionError("unexpected EOF")
+        data.extend(chunk)
+    return bytes(data)
+
+
+def varint(value):
+    result = bytearray()
+    while value >= 128:
+        result.append((value & 127) | 128)
+        value >>= 7
+    result.append(value)
+    return bytes(result)
+
+
+def envelope_ping(nonce):
+    ping = b"\x08" + varint(nonce)
+    return b"\x08\x2a\x62" + varint(len(ping)) + ping
+
+
+def packet(payload, kind=1):
+    return struct.pack("!IIHIB", MAGIC, len(payload) + 7, kind, 7, 1) + payload
+
+
+def packet_pong(nonce):
+    pong = b"\x08" + varint(nonce)
+    return packet(b"\x08\x2a\x6a" + varint(len(pong)) + pong, 2)
+
+
+def read_tcp_packet(peer):
+    prefix = exact(peer, 8)
+    magic, length = struct.unpack("!II", prefix)
+    return prefix + exact(peer, length) if magic == MAGIC else b""
+
+
+def upgrade(port, origin=ORIGIN):
+    peer = socket.create_connection(("127.0.0.1", port), 5)
+    peer.settimeout(8)
+    key = base64.b64encode(os.urandom(16)).decode()
+    request = ("GET /ws HTTP/1.1\r\n"
+               f"Host: 127.0.0.1:{port}\r\n"
+               "Upgrade: websocket\r\nConnection: Upgrade\r\n"
+               f"Sec-WebSocket-Key: {key}\r\n"
+               "Sec-WebSocket-Version: 13\r\n"
+               f"Origin: {origin}\r\n\r\n")
+    peer.sendall(request.encode())
+    reply = bytearray()
+    while not reply.endswith(b"\r\n\r\n"):
+        reply.extend(exact(peer, 1))
+        assert len(reply) < 8192
+    return peer, bytes(reply)
+
+
+def send_frame(peer, opcode, payload):
+    header = bytearray([0x80 | opcode])
+    length = len(payload)
+    if length < 126:
+        header.append(0x80 | length)
+    elif length <= 65535:
+        header.append(0x80 | 126)
+        header.extend(struct.pack("!H", length))
+    else:
+        header.append(0x80 | 127)
+        header.extend(struct.pack("!Q", length))
+    mask = os.urandom(4)
+    header.extend(mask)
+    peer.sendall(header + bytes(value ^ mask[i % 4]
+                                 for i, value in enumerate(payload)))
+
+
+def read_frame(peer):
+    first, second = exact(peer, 2)
+    length = second & 127
+    if length == 126:
+        length = struct.unpack("!H", exact(peer, 2))[0]
+    elif length == 127:
+        length = struct.unpack("!Q", exact(peer, 8))[0]
+    assert not second & 0x80, "server frame must be unmasked"
+    assert length <= 1024 * 1024
+    return first & 15, exact(peer, length)
+
+
+def close_code(peer):
+    while True:
+        opcode, payload = read_frame(peer)
+        if opcode == 8:
+            assert len(payload) >= 2
+            code = struct.unpack("!H", payload[:2])[0]
+            try:
+                send_frame(peer, 8, payload[:2])
+            except OSError:
+                pass  # The server may close TCP after sending its close code.
+            return code
+
+
+def run_host(binary, mode):
+    tcp_port, ws_port = reserve_port(), reserve_port()
+    while ws_port == tcp_port:
+        ws_port = reserve_port()
+    log = tempfile.NamedTemporaryFile(prefix="arena-ws-", suffix=".log",
+                                      delete=False)
+    process = subprocess.Popen([binary, str(tcp_port), str(ws_port), ORIGIN, mode],
+                               stdin=subprocess.DEVNULL, stdout=log,
+                               stderr=log)
+    def content():
+        log.flush()
+        with open(log.name, errors="replace") as stream:
+            return stream.read()
+    deadline = time.monotonic() + 8
+    while "WS_TEST_READY" not in content():
+        if process.poll() is not None or time.monotonic() > deadline:
+            raise AssertionError("WebSocket test host failed: " + content())
+        time.sleep(.02)
+    return process, log, content, tcp_port, ws_port
+
+
+def stop_host(process, log):
+    if process.poll() is None:
+        process.send_signal(signal.SIGUSR1)
+        try:
+            assert process.wait(timeout=8) == 0
+        except subprocess.TimeoutExpired:
+            process.kill()
+            process.wait()
+            raise
+    log.close()
+
+
+def run(binary):
+    process, log, content, tcp_port, ws_port = run_host(binary, "handshake")
+    try:
+        with socket.create_connection(("127.0.0.1", tcp_port), 5) as peer:
+            peer.settimeout(5)
+            peer.sendall(packet(envelope_ping(123)))
+            assert read_tcp_packet(peer) == packet_pong(123)
+        peer, response = upgrade(ws_port)
+        with peer:
+            assert response.startswith(b"HTTP/1.1 101")
+            send_frame(peer, 2, packet(envelope_ping(456)))
+            assert read_frame(peer) == (2, packet_pong(456))
+            send_frame(peer, 8, struct.pack("!H", 1000))
+            assert close_code(peer) == 1000
+        for origin in ("https://untrusted.example", ""):
+            denied, response = upgrade(ws_port, origin)
+            with denied:
+                assert response.startswith(b"HTTP/1.1 403")
+        peer, response = upgrade(ws_port)
+        with peer:
+            assert response.startswith(b"HTTP/1.1 101")
+            send_frame(peer, 1, b"text not allowed")
+            assert close_code(peer) == 1003
+        peer, response = upgrade(ws_port)
+        with peer:
+            send_frame(peer, 2, b"x" * (256 * 1024 + 9))
+            assert close_code(peer) == 1009
+        peer, response = upgrade(ws_port)
+        with peer:
+            send_frame(peer, 8, struct.pack("!H", 4001))
+            assert close_code(peer) == 4001
+        deadline = time.monotonic() + 3
+        while "code=4001" not in content() and time.monotonic() < deadline:
+            time.sleep(.02)
+        output = content()
+        assert "code=1003" in output and "code=4001" in output, output
+        assert "code=1009" in output, output
+    finally:

+        os.unlink(log.name)

+        assert match and int(match.group(1)) <= 1024 * 1024, content()
+    finally:
+        stop_host(process, log)
+        os.unlink(log.name)
+    print("TCP/WS pre-login, origin, size, close-code and queue limits passed")
