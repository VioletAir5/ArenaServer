+void AsioConnect::send(std::span<const std::byte> bytes)
+{
+    requireOwner();
+    if (m_closed || m_shutdownRequested || bytes.empty())
+    {
+        return;
+    }
+    auto keepAlive = weak_from_this().lock();
+    if (bytes.size() > kHardOutputLimit - m_pendingBytes)
+    {
+        std::cerr << "AsioConnect output hard limit reached\n";
+        close();
+        return;
+    }
+    // Each operation captures its own immutable buffer. Appending or
+    // closing the queue cannot invalidate memory still used by a
+    // cancelled operation. Coalesce small sends into a tail not used
+    // by the active write. This bounds per-chunk metadata as well as
+    // payload memory for tiny sends.
+    if (!m_output.empty() && (!m_writing || m_output.size() > 1) &&
+        m_output.back()->size() + bytes.size() <= 64 * 1024)
+    {
+        auto& tail = *m_output.back();
+        tail.insert(tail.end(), bytes.begin(), bytes.end());
+    }
+    else
+    {
+        auto buffer = std::make_shared<WriteBuffer>();
+        buffer->reserve(std::max<std::size_t>(8192, bytes.size()));
+        buffer->assign(bytes.begin(), bytes.end());
+        m_output.push_back(std::move(buffer));
+    }
+    m_pendingBytes += bytes.size();
+    updateWatermark();
+    writeNext();
+}

+void AsioConnect::close()
+{
+    requireOwner();
+    if (m_closed)
+    {
+        return;
+    }
+    auto keepAlive = weak_from_this().lock();
+    m_closed = true;
+    boost::system::error_code ignored;
+    m_socket.close(ignored);
+    m_idleTimer.cancel(ignored);
+    m_slowTimer.cancel(ignored);
+    m_output.clear();

+    m_readReady = false;
+    // Completion handlers retain buffers, never the connection.
+    // Notify once; the server defers map erasure until this dispatch
+    // has returned.

+    {
+        auto callback = m_closeCallback;
+        try
+        {
+            callback(this);
+        }
+        catch (const std::exception& e)
+        {
+            std::cerr << "AsioConnect close callback: " << e.what()
+                      << '\n';
+        }
+        catch (...)
+        {
+            std::cerr << "AsioConnect close callback failed\n";
+        }
+    }

+void AsioConnect::readNext()
+{
+    if (!m_started || m_closed || m_peerHalfClosed || m_readPaused ||
+        m_reading || m_readReady || m_delivering)
+    {
+        return;
+    }
+    m_reading = true;

+            if (!self || self->m_closed)
+            {
+                return;

+            self->m_reading = false;
+            self->m_readBytes = bytes;
+            self->m_readError = error;
+            self->m_readReady = true;
+            if (bytes != 0)
+            {
+                self->m_lastActivity = Clock::now();
+            }
+            if (!self->m_readPaused)
+            {
+                self->deliverRead();
+            }
+        }

+void AsioConnect::deliverRead()
+{
+    if (m_closed || !m_readReady || m_readPaused || m_delivering)
+    {
+        return;
+    }
+    const auto bytes = m_readBytes;
+    const auto error = m_readError;
+    m_readReady = false;
+    m_delivering = true;
+    if (bytes != 0 && m_messageCallback)
+    {
+        auto callback = m_messageCallback;
+        try
+        {
+            callback(
+                *this,
+                std::span<const std::byte>(m_input->data(), bytes)
+            );
+        }
+        catch (const std::exception& e)
+        {
+            std::cerr << "AsioConnect message callback: " << e.what()
+                      << '\n';
+            close();
+        }
+        catch (...)
+        {
+            close();
+        }
+    }
+    m_delivering = false;
+    if (m_closed)
+    {
+        return;
+    }
+    if (error == boost::asio::error::eof)
+    {
+        m_peerHalfClosed = true;
+        finishWrite();
+    }
+    else if (error)
+    {
+        ioError(error);
+    }
+    else
+    {
+        readNext();
+    }

+void AsioConnect::writeNext()
+{
+    if (!m_started || m_closed || m_writing || m_output.empty())
+    {
+        return;
+    }
+    m_writing = true;
+    auto buffer = m_output.front();
+    const auto offset = m_frontOffset;
+    m_socket.async_write_some(
+        boost::asio::buffer(
+            buffer->data() + offset,
+            buffer->size() - offset
+        ),
+        [weak = weak_from_this(),
+         buffer](boost::system::error_code error, std::size_t bytes)
+        {
+            auto self = weak.lock();
+            if (!self || self->m_closed)
+            {
+                return;
+            }
+            self->m_writing = false;
+            self->m_frontOffset += bytes;
+            self->m_pendingBytes -= bytes;
+            if (bytes != 0)
+            {
+                self->m_lastActivity = Clock::now();
+            }
+            if (self->m_frontOffset == buffer->size())
+            {
+                self->m_output.pop_front();
+                self->m_frontOffset = 0;
+            }
+            if (error)
+            {
+                self->ioError(error);
+                return;
+            }
+            self->updateWatermark();
+            self->writeNext();
+            self->finishWrite();
+        }
+    );
+}

+void AsioConnect::finishWrite()
+{
+    if (m_closed || m_pendingBytes != 0)
+    {
+        return;
+    }
+    if (m_shutdownRequested && !m_writeHalfClosed)
+    {
+        boost::system::error_code error;
+        m_socket.shutdown(
+            boost::asio::ip::tcp::socket::shutdown_send,
+            error
+        );
+        if (error && error != boost::asio::error::not_connected)
+        {
+            ioError(error);
+            return;
+        }
+        m_writeHalfClosed = true;
+    }
+    if (m_peerHalfClosed)
+    {
+        close();
+    }
+}

+void AsioConnect::updateWatermark()
+{
+    if (m_closed)
+    {
+        return;
+    }
+    WatermarkCallback callback;
+    if (!m_aboveHighWater && m_pendingBytes >= kHighWaterMark)
+    {
+        m_aboveHighWater = true;
+        m_highWaterSince = Clock::now();
+        const auto phase = m_highWaterSince;
+        m_slowTimer.expires_at(phase + kSlowClientTimeout);
+        m_slowTimer.async_wait(
+            [weak = weak_from_this(), phase](auto error)
+            {
+                auto self = weak.lock();
+                if (!error && self && !self->m_closed &&
+                    self->m_aboveHighWater &&
+                    self->m_highWaterSince == phase)
+                {
+                    std::cerr << "AsioConnect slow client timeout\n";
+                    self->close();
+                }
+            }
+        );
+        callback = m_highWaterCallback;
+    }
+    else if (m_aboveHighWater && m_pendingBytes <= kLowWaterMark)
+    {
+        m_aboveHighWater = false;
+        boost::system::error_code ignored;
+        m_slowTimer.cancel(ignored);
+        callback = m_lowWaterCallback;
+    }
+    if (callback)
+    {
+        try
+        {
+            callback(*this, m_pendingBytes);
+        }
+        catch (const std::exception& e)
+        {
+            std::cerr << "AsioConnect watermark callback: "
+                      << e.what() << '\n';
+            close();
+        }

+    // worker-internal descriptors don't appear as apparent connection

+    boost::asio::steady_timer prime;
+    std::optional<Guard> guard;
+    std::thread thread;

+        {
+            onRemoved(ticket);
+            return;
+        }
+        try
+        {
+            Tcp::socket socket(context);
+            Error error;
+            socket.assign(Tcp::v4(), accepted->fd, error);
+            if (error)

+                context,

+                idleTimeout

+                [](IConnection& c, auto bytes) { c.send(bytes); }

+                [](IConnection& c, auto) { c.pauseRead(); }
+            );
+            connection->setLowWaterMarkCallback(
+                [](IConnection& c, auto) { c.resumeRead(); }

+                {
+                    if (auto worker = weak.lock())
+                    {
+                        worker->deferRemove(ticket);
+                    }
+                }
+            );
+            connections.emplace(ticket, connection);
+            connection->start();
+            if (draining)
+            {
+                connection->shutdown();
+            }
+        }
+        catch (const std::exception& e)
+        {
+            std::cerr << "Asio connection setup failed: " << e.what()
+                      << '\n';

+            if (it != connections.end())
+            {
+                it->second->setCloseCallback({});
+                it->second->close();
+                connections.erase(it);
+            }
+            onRemoved(ticket);
+        }

+    {
+        if (cleaning)
+        {
+            return;
+        }

+        if (removalQueued)
+        {
+            return;
+        }
+        removalQueued = true;
+        boost::asio::post(
+            context,
+            [weak = weak_from_this()]
+            {
+                auto worker = weak.lock();
+                if (!worker || worker->cleaning)
+                {
+                    return;
+                }
+                worker->removalQueued = false;
+                auto pending = std::move(worker->removals);
+                worker->removals.clear();
+                for (auto ticket : pending)
+                {
+                    if (worker->connections.erase(ticket) != 0)
+                    {
+                        worker->onRemoved(ticket);
+                    }
+                }
+            }
+        );
+    }

+    void drain()
+    {
+        draining = true;
+        for (const auto& [ticket, connection] : connections)
+        {
+            (void)ticket;
+            connection->shutdown();
+        }
+    }
+
+    void forceClose()
+    {
+        forceClosing = true;
+        for (const auto& [ticket, connection] : connections)
+        {
+            (void)ticket;
+            connection->close();
+        }
+    }
+
+    void cleanup()
+    {
+        cleaning = true;
+        for (const auto& [ticket, connection] : connections)
+        {
+            (void)ticket;
+            connection->setCloseCallback({});
+            connection->close();
+        }
+        connections.clear();
+        removals.clear();

+        // abandon ready handlers, retaining buffers until context
+        // destruction.
+    }

+                        F_GETFD

+                                         F_SETFD,

+                            boost::system::generic_category()

+                    // allocation failure leaves the socket's RAII

+            return;

+        }
+    }

+    {
+        // Post all cleanups before waiting. Workers only post
+        // acknowledgements to base; they never wait for base, so
+        // joining cannot deadlock.
+        std::vector<std::future<void>> finished;
+        for (const auto& worker : workers)
+        {
+            if (!worker->thread.joinable())
+            {
+                worker->cleanup();
+                continue;
+            }
+            auto done = std::make_shared<std::promise<void>>();
+            finished.push_back(done->get_future());
+            worker->dispatch(
+                [worker, done]
+                {
+                    worker->cleanup();
+                    done->set_value();
+                }
+            );
+        }
+        for (auto& completion : finished)
+        {
+            completion.get();
+        }
+        for (const auto& worker : workers)
+        {
+            if (worker->thread.joinable())
+            {
+                worker->thread.join();
+            }
+        }
+        workers.clear();
+    }
+

+    {
+        if (state != State::Draining || !admissions.empty())
+        {
+            return;
+        }
+        Error ignored;
+        stopTimer.cancel(ignored);
+        cleanupWorkers();
+        state = State::Stopped;
+        baseGuard.reset();
+        std::cerr << "AsioTcpServer stopped\n";
+    }
+

+    {

+        {
+            std::terminate();
+        }
+        destroying = true;

+        Error ignored;
+        stopTimer.cancel(ignored);
+        cleanupWorkers();
+        admissions.clear();
+        state = State::Stopped;
+        baseGuard.reset();
+    }
